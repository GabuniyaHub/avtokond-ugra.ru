# Редирект HTTP -> HTTPS
server {
    listen 80;
    listen [::]:80;
    server_name avtokond-ugra.ru www.avtokond-ugra.ru;

    # Простейший редирект на HTTPS
    return 301 https://$host$request_uri;
}

# Основной HTTPS сервер
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name avtokond-ugra.ru www.avtokond-ugra.ru;

    root /var/www/avtokond/public;
    index index.html index.htm;

    # Раздача статики и SPA fallback
    location / {
        try_files $uri /index.html;
    }

    # Проксирование всего, что идёт на /api/, на бекенд на localhost:4080
    location /api/ {
        proxy_pass http://127.0.0.1:4080/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # SSL сертификаты от Certbot
    ssl_certificate /etc/letsencrypt/live/avtokond-ugra.ru/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/avtokond-ugra.ru/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Логи
    access_log /var/log/nginx/avtokond.access.log;
    error_log /var/log/nginx/avtokond.error.log;
}
